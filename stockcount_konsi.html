StockCount Pro • Offline-First LocalStorage & Google Sheets Integration

Kode Google Apps Script (`Code.gs`) untuk Tab `gid=243038121`

  1. Buka file Google Spreadsheet Anda, lalu klik menu Extensions (Ekstensi) > Apps Script.
  2. Hapus kode bawaan dan tempelkan seluruh kode di bawah ini (script otomatis mencari tab dengan gid=243038121).
  3. Klik Deploy (Terapkan) > New deployment (Deployment baru) > pilih tipe Web app.
  4. Setel Execute as: Me, dan Who has access: Anyone (Siapa saja), lalu klik Deploy.
  5. Salin Web App URL yang muncul dan tempelkan pada pengaturan di aplikasi ini.
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var targetGid = 243038121;
    var sheets = ss.getSheets();
    var sheet = null;

    // Cari tab sheet sesuai gid=243038121
    for (var i = 0; i < sheets.length; i++) {
      if (sheets[i].getSheetId() == targetGid) {
        sheet = sheets[i];
        break;
      }
    }
    if (!sheet) {
      sheet = ss.getActiveSheet();
    }

    var payload = JSON.parse(e.postData.contents);
    var items = payload.items || [];

    // Buat header otomatis jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        'Waktu_Kirim',
        'Sesi_Lokasi',
        'Petugas',
        'Barcode_SKU',
        'Nama_Produk',
        'Kategori',
        'Satuan',
        'Stok_Sistem',
        'Stok_Fisik',
        'Selisih',
        'Status',
        'Catatan'
      ]);
    }

    var timestamp = new Date();
    var rows = items.map(function(item) {
      return [
        timestamp,
        payload.sessionName || '',
        payload.pic || '',
        "'" + item.sku, // Tanda kutip agar barcode angka panjang tidak berubah format
        item.name || '',
        item.category || '',
        item.unit || 'Pcs',
        Number(item.systemQty) || 0,
        Number(item.countedQty) || 0,
        Number(item.variance) || 0,
        item.status || '',
        item.note || ''
      ];
    });

    if (rows.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, rows[0].length).setValues(rows);
    }

    return ContentService
      .createTextOutput(JSON.stringify({ status: 'success', inserted: rows.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
